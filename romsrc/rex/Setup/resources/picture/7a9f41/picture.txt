version 1

PackBitsRect bits1.png		# 96 x 22 pixels, black a set bit
	rowBytes auto		# 12: the width in bytes, rounded up to even
	bounds 8 352 30 448	# top left bottom right: the PNG's size
	srcRect 8 356 30 446
	dstRect 8 356 30 446
	mode srcCopy
	packing classic		# Apple's PackBits: runs of 3+ bytes, and a row's last two if equal
