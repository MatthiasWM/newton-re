# A QuickDraw picture as text: tools/pict/newtonpict.py (tools/pict/README.md).
# Rects are top left bottom right, points v h; `#` starts a comment.
picSize auto		# 0x0558: the picture's length, low 16 bits
picFrame 350 5 398 236
version 1
Clip 0 0 400 704		# the clip region: its rect, then any more of it in hex
PackBitsRect bits1.png		# 240 x 48 pixels, black a set bit
	rowBytes auto		# 30: the width in bytes, rounded up to even
	bounds 350 0 398 240	# top left bottom right: the PNG's size
	srcRect 350 5 398 236
	dstRect 350 5 398 236
	mode srcCopy
	packing classic		# Apple's PackBits: runs of 3+ bytes, and a row's last two if equal
OpEndPic
