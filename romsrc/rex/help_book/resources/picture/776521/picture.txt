# Rects are top left bottom right, points v h; `#` starts a comment.
picSize auto		# 0x008e: the picture's length, low 16 bits
picFrame -1 -1 12 11
version 2
HeaderOp -1 -1 -1 11 12 0		# version (-1), the frame at 72 dpi as 16.16 left top right bottom, reserved
DefHilite
Clip -1 -1 12 11		# the clip region: its rect, then any more of it in hex
PnSize 0 0		# v h
ShortLine 641 352 0 0		# from v h, by dh dv
LongComment 182 00010001		# SetLineWidth: kind, data in hex (- for none)
Clip 0 0 10 9		# the clip region: its rect, then any more of it in hex
BitsRect bits1.png		# 16 x 10 pixels, black a set bit
	rowBytes auto		# 2: the width in bytes, rounded up to even
	bounds 14 8 24 24	# top left bottom right: the PNG's size
	srcRect 14 10 24 19
	dstRect 0 0 10 9
	mode srcCopy
OpEndPic
