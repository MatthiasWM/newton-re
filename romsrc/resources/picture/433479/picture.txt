# A QuickDraw picture as text: tools/pict/newtonpict.py (tools/pict/README.md).
# Rects are top left bottom right, points v h; `#` starts a comment.
picSize auto		# 0x048b: the picture's length, low 16 bits
picFrame 18 12 119 113
version 1
Clip 0 0 264 352		# the clip region: its rect, then any more of it in hex
PackBitsRect bits1.png		# 112 x 101 pixels, black a set bit
	rowBytes auto		# 14: the width in bytes, rounded up to even
	bounds 18 8 119 120	# top left bottom right: the PNG's size
	srcRect 18 12 119 113
	dstRect 18 12 119 113
	mode srcCopy
	packing classic		# Apple's PackBits: runs of 3+ bytes, and a row's last two if equal
OpEndPic
